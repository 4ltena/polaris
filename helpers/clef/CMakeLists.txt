cmake_minimum_required(VERSION 3.20)
project(polaris_clef_worker C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This worker build currently supports macOS only")
endif()
if(NOT IS_ABSOLUTE "${POLARIS_LLAMA_ARCHIVE}")
  message(FATAL_ERROR "POLARIS_LLAMA_ARCHIVE must name the pinned source archive")
endif()
# Each configure starts with an empty extraction directory. An existing
# source tree cannot add unverified files through upstream source globs.
string(RANDOM LENGTH 16 ALPHABET 0123456789abcdef source_generation)
set(extraction "${CMAKE_CURRENT_BINARY_DIR}/source-${source_generation}")
if(EXISTS "${extraction}")
  message(FATAL_ERROR "Source generation already exists")
endif()
file(MAKE_DIRECTORY "${extraction}")
set(snapshot "${extraction}/runtime-source.tar.gz")
configure_file("${POLARIS_LLAMA_ARCHIVE}" "${snapshot}" COPYONLY)
file(SHA256 "${snapshot}" archive_hash)
if(NOT archive_hash STREQUAL "1d4a987d9bd048f6e93107e838026438df70fbde237fc0c69c63d600512b4bd9")
  message(FATAL_ERROR "Pinned llama.cpp archive digest mismatch")
endif()
file(CHMOD "${snapshot}" PERMISSIONS OWNER_READ)
file(ARCHIVE_EXTRACT INPUT "${snapshot}" DESTINATION "${extraction}")
set(POLARIS_LLAMA_SOURCE "${extraction}/llama.cpp-620d2225cc436ab0b001e3200e590e3b286ef9ab")
# Source archives have no VCS metadata. Do not probe the parent repository.
set(CMAKE_DISABLE_FIND_PACKAGE_Git TRUE)
set(GIT_EXECUTABLE "" CACHE FILEPATH "VCS probes disabled for source archive" FORCE)
set(GIT_EXE "" CACHE FILEPATH "VCS probes disabled for source archive" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_NUMBER 0 CACHE STRING "Archive build" FORCE)
set(LLAMA_BUILD_COMMIT "620d2225" CACHE STRING "Pinned upstream source" FORCE)
set(GGML_BUILD_COMMIT "620d2225" CACHE STRING "Pinned upstream source" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${POLARIS_LLAMA_SOURCE}" upstream)

add_executable(polaris-clef-worker worker.cpp)
target_compile_features(polaris-clef-worker PRIVATE cxx_std_17)
target_include_directories(polaris-clef-worker SYSTEM PRIVATE "${POLARIS_LLAMA_SOURCE}/tools/server")
target_link_libraries(polaris-clef-worker PRIVATE server-context vendor::hash)
target_compile_options(polaris-clef-worker PRIVATE -Wall -Wextra -Werror)
